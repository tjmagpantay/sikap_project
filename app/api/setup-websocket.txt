==================================================
FUTURE ARCHITECTURE (VPS/Cloud with WebSocket)
==================================================

1. HYBRID SYSTEM:
   - Database notifications (for persistence)
   - WebSocket real-time push (for instant updates)
   - Polling as fallback (if WebSocket fails)

2. TECHNOLOGY STACK OPTIONS:

   OPTION A - Node.js WebSocket Server:
   - Socket.IO server on port 3001
   - PHP sends HTTP requests to Node.js
   - Node.js broadcasts to connected clients
   
   OPTION B - PHP Ratchet WebSocket:
   - Pure PHP WebSocket server
   - Integrated with existing codebase
   - ReactPHP/Ratchet library

3. RECOMMENDED: Node.js Socket.IO
   - More stable for production
   - Better browser compatibility
   - Easier to scale

// websocket-server/server.js
const express = require('express');
const http = require('http');
const socketIo = require('socket.io');
const cors = require('cors');

const app = express();
const server = http.createServer(app);
const io = socketIo(server, {
  cors: {
    origin: "https://yourdomain.com",
    methods: ["GET", "POST"]
  }
});

app.use(cors());
app.use(express.json());

// Store connected users
const connectedUsers = new Map();

// WebSocket connection handling
io.on('connection', (socket) => {
  console.log('User connected:', socket.id);
  
  // User authentication and room joining
  socket.on('join', (data) => {
    const { userId, userType } = data;
    socket.join(`${userType}-${userId}`);
    connectedUsers.set(socket.id, { userId, userType });
    console.log(`${userType} ${userId} joined room`);
  });
  
  // Handle disconnection
  socket.on('disconnect', () => {
    connectedUsers.delete(socket.id);
    console.log('User disconnected:', socket.id);
  });
});

// HTTP endpoint for PHP to push notifications
app.post('/push-notification', (req, res) => {
  const { userId, userType, notification } = req.body;
  
  // Send to specific user room
  io.to(`${userType}-${userId}`).emit('notification', notification);
  
  res.json({ success: true, message: 'Notification sent' });
});

// Health check endpoint
app.get('/health', (req, res) => {
  res.json({ status: 'WebSocket server running', timestamp: new Date() });
});

const PORT = process.env.PORT || 3001;
server.listen(PORT, () => {
  console.log(`WebSocket server running on port ${PORT}`);
});

// In NotificationService.php - pushToSocket() method
private function pushToSocket($userId, $userType, $notificationData) {
  if (!$this->isWebSocketEnabled()) {
    return false; // Skip if WebSocket not available
  }
  
  try {
    $payload = [
      'userId' => $userId,
      'userType' => $userType,
      'notification' => $notificationData
    ];
    
    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, 'http://localhost:3001/push-notification');
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($payload));
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_TIMEOUT, 5);
    
    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    
    if ($httpCode === 200) {
      error_log("✅ WebSocket notification pushed for {$userType} {$userId}");
      return true;
    } else {
      error_log("❌ WebSocket push failed: HTTP {$httpCode}");
      return false;
    }
  } catch (Exception $e) {
    error_log("❌ WebSocket push error: " . $e->getMessage());
    return false;
  }
}

// Add to navbar JavaScript
class NotificationManager {
  constructor() {
    this.socket = null;
    this.pollingInterval = null;
    this.userId = null;
    this.userType = null;
  }
  
  init(userId, userType) {
    this.userId = userId;
    this.userType = userType;
    
    // Try WebSocket first, fallback to polling
    if (this.initWebSocket()) {
      console.log('✅ Using WebSocket notifications');
    } else {
      console.log('📡 Falling back to polling notifications');
      this.startPolling();
    }
  }
  
  initWebSocket() {
    try {
      // Only try WebSocket in production/VPS environment
      if (!window.location.hostname.includes('localhost') && 
          !window.location.hostname.includes('127.0.0.1')) {
        
        this.socket = io('wss://yourdomain.com:3001');
        
        this.socket.on('connect', () => {
          console.log('🔌 WebSocket connected');
          this.socket.emit('join', {
            userId: this.userId,
            userType: this.userType
          });
          this.stopPolling(); // Stop polling when WebSocket works
        });
        
        this.socket.on('notification', (notification) => {
          console.log('🔔 Real-time notification received:', notification);
          this.handleNewNotification(notification);
        });
        
        this.socket.on('disconnect', () => {
          console.log('🔌 WebSocket disconnected, starting polling fallback');
          this.startPolling();
        });
        
        return true;
      }
    } catch (error) {
      console.log('❌ WebSocket failed, using polling:', error);
      return false;
    }
    return false;
  }
  
  startPolling() {
    this.pollingInterval = setInterval(() => {
      this.fetchNotifications();
    }, 15000);
  }
  
  stopPolling() {
    if (this.pollingInterval) {
      clearInterval(this.pollingInterval);
      this.pollingInterval = null;
    }
  }
  
  handleNewNotification(notification) {
    // Update UI immediately
    this.notifications.unshift(notification);
    this.unreadCount++;
    
    // Show toast notification
    this.showToast(notification.title, notification.message);
  }
}

/**
 * Enhanced notification creation with WebSocket support
 * This method ensures all notifications go through a single integration point
 */
private function createAndPushNotification($userId, $userType, $type, $title, $message, $link = null, $data = null)
{
    try {
        error_log("🔔 Creating notification for {$userType} {$userId}: {$title}");
        
        // 1. Always save to database (for persistence and polling)
        $databaseResult = $this->notification->create($userId, $userType, $type, $title, $message, $link, $data);
        
        if (!$databaseResult) {
            error_log("❌ Failed to save notification to database");
            return false;
        }
        
        // 2. Try to push via WebSocket (if available)
        $notificationData = [
            'notification_id' => $this->notification->getLastInsertId(),
            'user_id' => $userId,
            'user_type' => $userType,
            'type' => $type,
            'title' => $title,
            'message' => $message,
            'link' => $link,
            'data' => $data,
            'status' => 'unread',
            'created_at' => date('Y-m-d H:i:s')
        ];
        
        $websocketResult = $this->pushToSocket($userId, $userType, $notificationData);
        
        if ($websocketResult) {
            error_log("✅ Notification sent via WebSocket and database");
        } else {
            error_log("📡 Notification saved to database (WebSocket not available - using polling)");
        }
        
        return $databaseResult;
        
    } catch (Exception $e) {
        error_log("❌ Error creating notification: " . $e->getMessage());
        return false;
    }
}
